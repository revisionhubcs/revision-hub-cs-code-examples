Word <- "Computer"
OUTPUT LENGTH(Word)              // 8
OUTPUT SUBSTRING(Word, 4, 3)     // "put"
OUTPUT UCASE(Word)               // "COMPUTER"
OUTPUT LCASE(Word)               // "computer"
