PROCEDURE ShowLine(Length : INTEGER)
   FOR Count <- 1 TO Length
      OUTPUT "-"
   NEXT Count
ENDPROCEDURE

FUNCTION AreaOfRectangle(Width : INTEGER, Height : INTEGER) RETURNS INTEGER
   RETURN Width * Height
ENDFUNCTION

CALL ShowLine(10)
OUTPUT AreaOfRectangle(4, 5)
