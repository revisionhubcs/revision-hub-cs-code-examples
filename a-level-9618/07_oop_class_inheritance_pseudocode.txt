CLASS Account
   PRIVATE Owner : STRING
   PRIVATE Balance : REAL
   PUBLIC PROCEDURE NEW(GivenOwner : STRING)
      Owner <- GivenOwner
      Balance <- 0
   ENDPROCEDURE
   PUBLIC FUNCTION GetBalance() RETURNS REAL
      RETURN Balance
   ENDFUNCTION
ENDCLASS

CLASS SavingsAccount INHERITS Account
   PRIVATE Rate : REAL
ENDCLASS
